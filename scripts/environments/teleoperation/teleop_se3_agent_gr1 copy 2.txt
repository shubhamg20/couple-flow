@dataclass
class EpisodeData:
    """Data structure for episode information."""
    episode_num: int
    trajectory: List[Dict[str, Any]]
    initial_objects: Dict[str, Any]


class EpisodeSaver:
    """Handles asynchronous saving of episode data."""
    
    def __init__(self, task_name: str):
        self.task_name = task_name
        self.task_dir = Path(f"source/recorded_runs/{args_cli.robot}/{task_name}")
        self.task_dir.mkdir(parents=True, exist_ok=True)
        
        self.save_queue = queue.Queue()
        self.is_running = True
        self.saver_thread = threading.Thread(target=self._save_worker, daemon=True)
        self.saver_thread.start()
    
    def _save_worker(self):
        """Worker thread for saving episodes."""
        while self.is_running:
            try:
                episode_data = self.save_queue.get(timeout=1.0)
                if episode_data is None:  # Poison pill
                    break
                self._save_episode_to_disk(episode_data)
                print(f"[RECORDING] Episode {episode_data.episode_num} saved.")
                self.save_queue.task_done()
            except queue.Empty:
                continue
    
    def _save_episode_to_disk(self, episode_data: EpisodeData):
        """Save episode data to disk - convert ALL torch tensors to numpy."""
        def convert_to_numpy(obj):
            """Recursively convert torch tensors to numpy arrays."""
            if isinstance(obj, torch.Tensor):
                return obj.cpu().detach().numpy()
            elif isinstance(obj, dict):
                return {k: convert_to_numpy(v) for k, v in obj.items()}
            elif isinstance(obj, list):
                return [convert_to_numpy(item) for item in obj]
            else:
                return obj
        
        # Convert all tensors in trajectory to numpy
        trajectory_numpy = convert_to_numpy(episode_data.trajectory)
        initial_objects_numpy = convert_to_numpy(episode_data.initial_objects)
        
        episode_file = self.task_dir / f"episode{episode_data.episode_num}.pkl"
        with open(episode_file, 'wb') as f:
            pickle.dump({
                "episode": episode_data.episode_num,
                "trajectory": trajectory_numpy,
                "initial_objects": initial_objects_numpy
            }, f)
        
        print(f"[RECORDING] Episode {episode_data.episode_num} SAVED to {episode_file} ({len(episode_data.trajectory)} timesteps)")
    
    def queue_episode(self, episode_data: EpisodeData) -> bool:
        """Queue an episode for saving."""
        if len(episode_data.trajectory) < 10:
            return False
        print(f"[RECORDING] Queuing Episode {episode_data.episode_num} for saving ({len(episode_data.trajectory)} timesteps)")
        self.save_queue.put(episode_data)
        return True
    
    def shutdown(self):
        """Shutdown the saver thread."""
        self.is_running = False
        self.save_queue.put(None)  # Poison pill
        self.saver_thread.join(timeout=5.0)


class RecordingManager:
    """Manages trajectory recording state and operations."""
    
    def __init__(self, task_name: str, enabled: bool = False):
        self.enabled = enabled
        self.task_name = task_name
        self.current_episode = 0
        self.current_trajectory = []
        self.episode_initial_objects = {}
        self.timestep = 0
        self.teleoperation_active = False
        
        if self.enabled:
            self.current_episode = self._get_next_episode_number()
            self.saver = EpisodeSaver(task_name)
            print("\n" + "="*60)
            print("TRAJECTORY RECORDING ENABLED")
            print(f"Task: {task_name}")
            print(f"Starting Episode: {self.current_episode}")
            print("="*60 + "\n")
        else:
            self.saver = None
    
    def _get_next_episode_number(self) -> int:
        """Find the next episode number by checking existing episodes in task folder."""
        task_dir = Path(f"source/recorded_runs/{args_cli.robot}/{self.task_name}")

        if not task_dir.exists():
            return 0
        
        episode_files = list(task_dir.glob("episode*.pkl"))
        if not episode_files:
            return 0
        
        episode_numbers = []
        for f in episode_files:
            try:
                num = int(f.stem.replace("episode", ""))
                episode_numbers.append(num)
            except ValueError:
                pass
        return max(episode_numbers) + 1 if episode_numbers else 0
    
    def start_recording(self):
        """Start recording trajectory."""
        self.teleoperation_active = True
        self.timestep = 0
        self.current_trajectory = []
        self.episode_initial_objects = {}
        if self.enabled:
            print(f"[RECORDING] Starting Episode {self.current_episode}")
    
    def stop_recording(self):
        """Stop recording trajectory."""
        self.teleoperation_active = False
    
    def add_step(self, step_data: Dict[str, Any]):
        """Add a step to the current trajectory."""
        if self.enabled and self.teleoperation_active:
            self.current_trajectory.append(step_data)
            self.timestep += 1
    
    def save_and_reset(self) -> bool:
        """Save current episode and reset for next episode."""
        if not self.enabled or not self.saver:
            return False
        
        episode_data = EpisodeData(
            episode_num=self.current_episode,
            trajectory=self.current_trajectory,
            initial_objects=self.episode_initial_objects
        )
        saved = self.saver.queue_episode(episode_data)
        if saved:
            self.current_episode += 1
            self.current_trajectory = []
            self.episode_initial_objects = {}
            self.timestep = 0
            print(f"[RECORDING] Episode queued for saving. Starting Episode {self.current_episode}")
        return saved
    
    def delete_current_episode(self):
        """Delete the current episode data."""
        self.current_trajectory = []
        self.episode_initial_objects = {}
        self.timestep = 0
        print(f"[RECORDING] Episode {self.current_episode} deleted.")
    
    def shutdown(self):
        """Shutdown the recording manager."""
        if self.saver:
            self.saver.shutdown()


class TeleoperationCallbacks:
    """Manages teleoperation callbacks and state."""
    
    def __init__(self, env, recording_manager: RecordingManager, visionpro_client: Optional[VisionProClient] = None):
        self.env = env
        self.recording_manager = recording_manager
        self.visionpro_client = visionpro_client
        self.should_reset_env = False
    
    def reset_environment(self):
        """Trigger environment reset."""
        self.recording_manager.stop_recording()
        self.should_reset_env = True
        self.recording_manager.delete_current_episode()
        if self.visionpro_client:
            self.visionpro_client.send_command("Resetting Teleoperation")
        print("Reset triggered - Environment will reset on next step")
    
    def start_teleoperation(self):
        """Start teleoperation."""
        self.recording_manager.start_recording()
        if self.visionpro_client:
            self.visionpro_client.send_command("Starting Teleoperation")
        print("🤓"*12 + "Teleoperation Activated" + "🤓"*12)
    
    def stop_teleoperation(self):
        """Stop teleoperation."""
        self.recording_manager.stop_recording()
        if self.visionpro_client:
            self.visionpro_client.send_command("Stopping Teleoperation")
        saved = self.recording_manager.save_and_reset()
        if saved:
            print("👺"*12 + "Teleoperation Deactivated" + "👺"*12)
    
    def save_and_reset_teleoperation(self):
        """Save current episode and reset."""
        if self.visionpro_client:
            self.visionpro_client.send_command("Saving and Resetting Teleoperation")
        print("🥳"*12 + "Save & Reset triggered" + "🥳"*12)
        
        saved = self.recording_manager.save_and_reset()
        return saved
    
    def get_callbacks_dict(self) -> Dict[str, Callable]:
        """Get dictionary of callbacks for teleop device."""
        return {
            "RESET": self.reset_environment,
            "PLAY": self.start_teleoperation,
            "START": self.start_teleoperation,
            "STOP": self.stop_teleoperation,
            "SAVE": self.save_and_reset_teleoperation,
        }


class OculusTeleop:
    """Oculus teleoperation device for SE(3) commands."""
    
    def __init__(self, env, pos_sensitivity: float = 1, rot_sensitivity: float = 1, robot_type: str = "franka"):
        self.pos_sensitivity = pos_sensitivity
        self.rot_sensitivity = rot_sensitivity
        self.robot_type = robot_type
        self.env = env
        self._callbacks = {}
        self._step_count = 0
        self._current_command = None
        self.action_dim = 7
        self.controller = VRPolicy()

    def reset(self) -> None:
        """Reset the device."""
        self._step_count = 0
        self._current_command = None
        self.controller.reset_state()

    def add_callback(self, key: str, callback) -> None:
        """Add callback."""
        self._callbacks[key] = callback
    
    def advance(self) -> torch.Tensor:
        if hasattr(self.controller, '_state') and self.controller._state.get('movement_enabled', False):
            self._current_command = self._generate_franka_command()
        else:
            self._current_command = np.zeros(7)
        self._step_count += 1 
        return torch.tensor(self._current_command, dtype=torch.float32)

    def _generate_franka_command(self) -> np.ndarray:
        eef_idx = self.env.scene["robot"].data.body_names.index("panda_hand")
        eef_pos = self.env.scene["robot"].data.body_pos_w[0, eef_idx].cpu().detach().numpy()
        eef_quat = self.env.scene["robot"].data.body_quat_w[0, eef_idx] 

        joint_names = self.env.scene["robot"].data.joint_names
        joint_positions = self.env.scene["robot"].data.joint_pos[0]        
        gripper_indices = [
            joint_names.index("panda_finger_joint1"),
            joint_names.index("panda_finger_joint2"),
        ]
        gripper_state = joint_positions[gripper_indices]

        robot_base_idx = self.env.scene["robot"].data.body_names.index("panda_link0")
        robot_base_quat = self.env.scene["robot"].data.body_quat_w[0, robot_base_idx][[1, 2, 3, 0]].cpu().detach().numpy()
        robot_base_rot = R.from_quat(robot_base_quat)
        robot_base_rot_inv = robot_base_rot.inv()

        eef_quat = eef_quat[[1, 2, 3, 0]].cpu().detach().numpy()
        eef_rot = R.from_quat(eef_quat)
        eef_rot_local = robot_base_rot_inv * eef_rot
        eef_euler = eef_rot_local.as_euler('xyz')

        robot_base_pos = self.env.scene["robot"].data.body_pos_w[0, robot_base_idx].cpu().detach().numpy()
        eef_pos_local = robot_base_rot_inv.apply(eef_pos - robot_base_pos)
        
        cartesian_position = np.concatenate([eef_pos_local, eef_euler])
        gripper_position = gripper_state.mean().cpu().detach().numpy()
        robot_state = {
            "cartesian_position": cartesian_position,
            "gripper_position": gripper_position
        }
        action, controller_action_info = self.controller.forward({"robot_state": robot_state}, include_info=True)
        return action

    def __str__(self) -> str:
        return f"OculusTeleop(robot={self.robot_type})"

def get_image(env) -> torch.Tensor:
    """Get image from camera - returns torch tensor (defer .cpu() until saving)."""
    rgb_data = env.scene["tiled_camera"].data.output["rgb"]
    return rgb_data[0].clone()  # Return torch tensor directly


def extract_trajectory_data(env, args_cli, timestep, action):
    """Extract current state for trajectory recording - keep as torch tensors."""
    if args_cli.robot == "franka":
        eef_idx = env.scene["robot"].data.body_names.index("panda_hand")
        eef_pos_w = env.scene["robot"].data.body_pos_w[0, eef_idx]
        eef_quat_w = env.scene["robot"].data.body_quat_w[0, eef_idx][[1, 2, 3, 0]]
        
        joint_names = env.scene["robot"].data.joint_names
        joint_positions = env.scene["robot"].data.joint_pos[0]
        gripper_indices = [joint_names.index("panda_finger_joint1"), joint_names.index("panda_finger_joint2")]
        gripper_state = joint_positions[gripper_indices].mean()

    elif args_cli.robot == "gr1t2":
        eef_pos_w = get_right_eef_pos(env)
        hand = get_hand_state(env).flatten()
        print("Hand state: [{}]".format(", ".join("{:.1f}".format(float(h)) for h in hand.cpu().numpy())))
        if (hand[5:10] < -0.6).sum() >= 4:
            gripper_state = torch.tensor(1.0, device=hand.device)  # Closed
        else:
            gripper_state = torch.tensor(0.0, device=hand.device)  # Open
    
    # Get camera parameters for projection - keep as tensors
    K = env.scene["tiled_camera"].data.intrinsic_matrices[0]
    q = env.scene["tiled_camera"].data.quat_w_world[0]
    t = env.scene["tiled_camera"].data.pos_w[0]
    
    # Store image as torch tensor
    img_tensor = get_image(env)
    
    objects = {}
    object_names = ["sushi", "mug", "apple"]
    for obj_name in object_names:
        if obj_name in env.scene.keys():
            obj_pos = env.scene[obj_name].data.body_pos_w[0][0]
            obj_quat = env.scene[obj_name].data.body_quat_w[0][0]
            objects[obj_name] = {
                "pos": obj_pos, 
                "quat": obj_quat
            }
    
    step_data = {
        "timestep": timestep,
        "franka_eef": {
            "pos": eef_pos_w,
            "quat": eef_quat_w if args_cli.robot == "franka" else None,
            "gripper": gripper_state
        },
        "image": img_tensor,
        "camera": {
            "intrinsics": K,
            "quat": q,
            "translation": t
        },
        "objects": objects
    }
    
    return step_data

def set_up_keyboard(env):
    input_interface = carb.input.acquire_input_interface()

    print("Keyboard reset (R) enabled.")

def main() -> None:
    """Run teleoperation with per-episode trajectory recording."""
    
    # Initialize recording manager
    recording_manager = RecordingManager(
        task_name=args_cli.task_name.lower(),
        enabled=args_cli.record_trajectory
    )
    
    # Initialize VisionPro client if using GR1T2
    visionpro_client = None
    if args_cli.robot == "gr1t2":
        visionpro_client = VisionProClient("192.168.0.50", 5555)
    
    # Set default task if not specified
    if args_cli.task is None:
        if args_cli.robot == "franka":
            args_cli.task = "Isaac-PickPlace-Franka-custom"
        elif args_cli.robot == "gr1t2":
            args_cli.task = "Isaac-PickPlace-GR1T2-Abs-v0"
        else:
            raise ValueError(f"No default task for robot type: {args_cli.robot}")
    
    print(f"Using robot: {args_cli.robot}")
    print(f"Using task: {args_cli.task}")
    
    # Configure environment
    env_cfg = parse_env_cfg(args_cli.task, device=args_cli.device, num_envs=args_cli.num_envs)
    env_cfg.env_name = args_cli.task
    env_cfg.sim.gravity = [0.0, 0.0, -9.81]
    env_cfg.terminations.time_out = None

    try:
        env = gym.make(args_cli.task, cfg=env_cfg).unwrapped
        if "Reach" in args_cli.task:
            omni.log.warn(
                f"The environment '{args_cli.task}' does not support gripper control. The device command will be ignored."
            )
    except Exception as e:
        omni.log.error(f"Failed to create environment: {e}")
        simulation_app.close()
        return
    
    set_up_keyboard(env)

    # Initialize callback manager
    callback_manager = TeleoperationCallbacks(env, recording_manager, visionpro_client)
    # Set initial teleoperation state
    if args_cli.teleop_device.lower() == "oculus":
        recording_manager.teleoperation_active = True

    # Create teleop interface
    teleop_interface = None
    try:
        if hasattr(env_cfg, "teleop_devices") and args_cli.teleop_device in env_cfg.teleop_devices.devices:
            teleop_interface = create_teleop_device(
                args_cli.teleop_device, env_cfg.teleop_devices.devices, callback_manager.get_callbacks_dict()
            )
        else:
            omni.log.warn(f"No teleop device '{args_cli.teleop_device}' found in environment config. Creating default.")
            sensitivity = args_cli.sensitivity
            if args_cli.teleop_device.lower() == "keyboard":
                teleop_interface = Se3Keyboard(
                    Se3KeyboardCfg(pos_sensitivity=0.05 * sensitivity, rot_sensitivity=0.05 * sensitivity)
                )
            elif args_cli.teleop_device.lower() == "spacemouse":
                teleop_interface = Se3SpaceMouse(
                    Se3SpaceMouseCfg(pos_sensitivity=0.05 * sensitivity, rot_sensitivity=0.05 * sensitivity)
                )
            elif args_cli.teleop_device.lower() == "gamepad":
                teleop_interface = Se3Gamepad(
                    Se3GamepadCfg(pos_sensitivity=0.1 * sensitivity, rot_sensitivity=0.1 * sensitivity)
                )
            elif args_cli.teleop_device.lower() == "oculus":
                teleop_interface = OculusTeleop(
                    env, 
                    pos_sensitivity=0.5 * sensitivity, 
                    rot_sensitivity=0.5 * sensitivity,
                    robot_type=args_cli.robot
                )
            else:
                omni.log.error(f"Unsupported teleop device: {args_cli.teleop_device}")
                env.close()
                simulation_app.close()
                return

            for key, callback in callback_manager.get_callbacks_dict().items():
                try:
                    teleop_interface.add_callback(key, callback)
                except (ValueError, TypeError) as e:
                    raise ValueError(f"Failed to add callback for key '{key}': {e}")
                
    except Exception as e:
        omni.log.error(f"Failed to create teleop device: {e}")
        env.close()
        simulation_app.close()
        return

    if teleop_interface is None:
        omni.log.error("Failed to create teleop interface")
        env.close()
        simulation_app.close()
        return
    
    print(f"Using teleop device: {teleop_interface}")
    # teleop_interface.reset()

    # Initialize robot
    if args_cli.robot == "franka":
        print("Initialising Franka joint position targets to current positions")
        current_joint_pos = env.scene["robot"].data.joint_pos.clone()
        env_ids = torch.arange(env.num_envs, device=env.device)
        root_state = env.scene["robot"].data.default_root_state.clone()
        env.scene["robot"].write_root_state_to_sim(root_state, env_ids=env_ids)
        env.scene["robot"].set_joint_position_target(current_joint_pos)
        env.sim.step(render=False)

    # Set camera view
    if args_cli.robot == "franka":
        env.sim.set_camera_view(eye=(-0.6, -0.3, 1.3), target=(1.1, 1.8, 0.9))
    else:
        env.sim.set_camera_view(eye=(0.0, 2.0, 2.0), target=(0.0, 0.0, 1.0))
    
    print("Teleoperation started.")
    
    record_frequency_hz = 20.0  
    record_interval = 1.0 / record_frequency_hz
    last_record_time = time.time()
    
    env.reset()
    teleop_interface.reset()
    
    # Main simulation loop
    while simulation_app.is_running():
        try:
            with torch.inference_mode():
                action = teleop_interface.advance()

                if recording_manager.teleoperation_active:
                    actions = action.repeat(env.num_envs, 1)
                    if args_cli.robot == "gr1t2":
                        env.step(actions[:36])
                    else: 
                        env.step(actions[:, :7])
                else:
                    env.sim.render()
                
                # Record trajectory if enabled
                now = time.time()
                if recording_manager.enabled and recording_manager.teleoperation_active and (now - last_record_time) >= record_interval:
                    step_data = extract_trajectory_data(env, args_cli, recording_manager.timestep, action)
                    recording_manager.add_step(step_data)
                    last_record_time = now
                    
                    # Capture initial object positions on first timestep of episode
                    if recording_manager.timestep == 1:  # Just added first step
                        object_names = ["sushi", "mug", "apple"]
                        for obj_name in object_names:
                            if obj_name in env.scene.keys():
                                obj_pos = env.scene[obj_name].data.body_pos_w[0][0].cpu().detach().numpy()
                                obj_quat = env.scene[obj_name].data.body_quat_w[0][0].cpu().detach().numpy()
                                recording_manager.episode_initial_objects[obj_name] = {
                                    "pos": obj_pos.tolist(), 
                                    "quat": obj_quat.tolist()
                                }
                
                if callback_manager.should_reset_env:
                    env.reset()
                    callback_manager.should_reset_env = False
                    print("Environment reset complete")

        except Exception as e:
            omni.log.error(f"Error during simulation step: {e}")
            break

    # Cleanup
    recording_manager.shutdown()
    env.close()
    print("Environment closed")


if __name__ == "__main__":
    try:
        main()
    finally:
        simulation_app.close()
